TFRKC_rework_prepare_partition_effect = {
 if = { limit = { TFRKC_rework_map_ready_trigger = yes NOT = { has_global_flag = TFRKC_rework_partition_complete } NOT = { has_global_flag = TFRKC_rework_actual_started } KOR = { exists = yes has_war = no } }
  TFRKC_rework_bound_partition_effect = yes
 }
}

# set TFRKC_rework_partition_complete. It never declares a war itself.

 TFRKC_rework_refresh_border_wars_effect = yes
 }
}
