KOR_PLP_influence_add = {

KOR_PLP_party_system_setup_effect = {
	hidden_effect = {
		# Clear Olds
		KOR_PLP_debug_clear_all_focus_effect = yes
		KOR_PLP_reset_now_focus_effect = yes

		# Setting Start
		set_country_flag = KOR_PLP_party_setup_flag

		# Start President Influence
		set_variable = { 
			KOR_PLP_president_faction_var = 1 # PP Faction
		}
		set_variable = {
			KOR_PLP_president_influence_var = 3
		}
		
		# Start Faction influence
		set_variable = {
			KOR_plp_pp_inf_var = 4
		}
		set_variable = {
			KOR_plp_jp_inf_var = 3
		}
		set_variable = {
			KOR_plp_lp_inf_var = 1
		}

		# Start Lead Faction
		set_variable = {
			PLPHeadFaction_var = 1 # Progressive Party
		}

		# Start Head of Party
		set_variable = {
			PLPPartyLeader_var = 2 # Sim Sang Jung
		}

		# Focuses
		KOR_PLP_debug_add_all_focus_effect = yes
		
		# Setup Starting Focus
		set_variable = { KOR_plp_focus_selected = 7 }
		set_country_flag = KOR_plp_set_net_focus_flag
		
		# Clamps
		KOR_PLP_party_influence_clamp = yes
		KOR_PLP_president_influence_clamp = yes
	}
}
