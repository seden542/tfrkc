add_namespace = ai_events

country_event = {
	id = ai_events.100
	hidden = yes

	trigger = {
		AND = {
			JAP = {
				is_faction_leader = yes
			}
			CHI = {
				has_war_with = PRC
				is_in_faction_with = JAP
			}
		}
	}

	fire_only_once = yes
	is_triggered_only = yes

	option = {
		log = "[GetLogInfo]: event ai_events.100 option 1"
		KOR = {
			country_event = { id = korea.53 days = 2 }
		}
	}
}
