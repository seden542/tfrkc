		}
		triggers = {
			GAW_card_button_click_enabled = {
				GAW_plan_isnt_ongoing_trigger = yes
				NOT = { check_variable = { var_GAW_card_1_select = TFRKP_GAW_card_have_array^idx } }
				NOT = { check_variable = { var_GAW_card_2_select = TFRKP_GAW_card_have_array^idx } }
			}
		}
		effects = {
			GAW_card_close_click = {
				clr_country_flag = GAW_card_select_1_flag
				clr_country_flag = GAW_card_select_2_flag
				clr_country_flag = TFRKP_show_card_select_flag
			}
			GAW_card_button_click = {
				set_temp_variable = { TFRKP_GAW_selected_card_temp = TFRKP_GAW_card_have_array^idx }
				if = {
					limit = {
						has_country_flag = GAW_card_select_1_flag
						check_variable = { var_GAW_card_1_select > 0 }
					}
					add_to_array = { TFRKP_GAW_card_have_array = var_GAW_card_1_select }
				}
				if = {
					limit = {
						has_country_flag = GAW_card_select_2_flag
						check_variable = { var_GAW_card_2_select > 0 }
					}
					add_to_array = { TFRKP_GAW_card_have_array = var_GAW_card_2_select }
				}
				if = {
					limit = {
						has_country_flag = GAW_card_select_1_flag
					}
					set_variable = { var_GAW_card_1_select = TFRKP_GAW_selected_card_temp }
				}
				if = {
					limit = {
						has_country_flag = GAW_card_select_2_flag
					}
					set_variable = { var_GAW_card_2_select = TFRKP_GAW_selected_card_temp }
				}
				remove_from_array = { array = TFRKP_GAW_card_have_array index = idx }
				add_to_variable = { TFRKP_GAW_card_have_dirty = 1 }
				clr_country_flag = GAW_card_select_1_flag
				clr_country_flag = GAW_card_select_2_flag
				clr_country_flag = TFRKP_show_card_select_flag
			}
		}
		properties = {
			GAW_card_button = {
				image = [GAW_card_select_image]
			}
		}
	}
