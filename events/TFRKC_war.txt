  log = "TFRKC_UNIFY_TRACE worker_enter actor=[ROOT.GetTag]"

   has_country_flag = TFRKC_unification_completion_queued

  }

   }
   else = {
    clr_country_flag = TFRKC_unification_pending
    log = "TFRKC_UNIFY_TRACE worker_cancelled_conditions_changed actor=[ROOT.GetTag]"
   }
  }
 }
